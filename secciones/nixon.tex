\begin{frame}{Ejemplo JMS para RTV: ejecución local}
\small
{\small\textbf{Evento propuesto:} \texttt{InspeccionIniciada}.}
\vspace{2mm}
\begin{enumerate}
  \item Terminal A: auditoría. Terminal B: indicadores.\\Ambos suscritos a \texttt{rtv.inspecciones.iniciada}.
  \item Terminal C: productor Java que publica un inicio ficticio.
  \item Ambos receptores muestran el mismo cuerpo y \texttt{JMSMessageID}.
\end{enumerate}
\begin{block}{Datos del ejemplo}
\small
\texttt{eventoId: evt-rtv-001}\quad\texttt{turnoId: 42}\par
\texttt{inspeccionId: identificador de la inspección}\par
\texttt{tipo: InspeccionIniciada}
\end{block}
{\small\textbf{Resultado observado:} un evento llegó a las dos suscripciones.}\par\vspace{2mm}
{\scriptsize\color{gris}Prueba local con Apache Artemis. Suscripciones no durables, activas antes del envío.}
\vfill\hfill{\tiny\color{gris}Fuente: \cite{jakarta2022}}\vspace{-0.11cm}
\end{frame}
